@testing:case="test_topic_class_variable_order"
try {undef(`sharedT, SHARED)} catch(ex) {}
share table(1:0, `sym`val, [STRING, INT]) as sharedT
go

a = 7
b = 8

class Data {
    a :: INT
    b :: DOUBLE
    sharedTInData :: ANY

    def Data() {
        a = 1
        b = 2.0
        sharedTInData  = sharedT
    }

    def getA(a) {
        return a
    }

    def getB() {
        return b
    }

    def insertT(sym, val) {    // attribute take precedence of shared variable
        insert into sharedTInData values(sym, val)
    }

    def getSharedT(sharedT) {    // function para take precedence of attribute and shared variable
        return sharedT
    }
}

d = Data()

d.insertT("1", 1)
d.insertT("2", 2)
d.insertT("3", 3)

assert 1, d.getA(4) == 4
assert 2, eqObj(d.getB(), 2.0)
assert 3, each(eqObj, table(symbol(["1", "2", "3"]) as symcol, [1, 2, 3] as valcol).values(), d.sharedTInData.values())
assert 4, d.getSharedT(10) == 10

@testing:case="test_topic_class_eval_in_function"
class Data {
    sym :: STRING
    val :: INT

    def Data(sym_, val_) {
        sym = sym_
        val = val_
    }

    def getVal() {
        return eval(<val + 1>)
    }

    def getValParams(a) {
        return 1 + eval(<val + a>)
    }
}

assert 2, eval(<d.getVal()>) == 2

        sym = sym_
        val = val_
    }

    }
