@testing:case="test_topic_class_inherit_with_DOUBLE[]"
class F1{
	id :: INT
	name :: STRING
	value :: DOUBLE[] VECTOR
	def F1(id_, name_){
		id ,name = id_,name_
	}	
	def setVal(value_){
		value = value_
	}
	def getFlag(){
		if(value.isNull().all()){
			return false
		}else{
			return true	
		}
	}
}

class F2 : F1{
	position  :: POINT
	val :: DOUBLE[] VECTOR
	def F2(position_){
		F1(int(1), string(1))
		position = position_
	}

	def getPosition(){
		return position	
	}
	def setVal2(id){
		position = point(id,id)
	}
	def setVal(val_){
		val = val_		
	}
}

f1 = F1(11, "ttdfa")
f = F2(point(1,1))
assert 1, f.id == 1
assert 2, f.name == "1"
assert 3,  f1.id == 11
assert 4, f1.name == "ttdfa"
f.setVal(arrayVector(take(1 3 5, 10).cumsum().int(), take(double(1..10), 28)))
assert 5, f.position == point(1,1)
assert 6, each(eqObj,f.val, arrayVector(take(1 3 5, 10).cumsum().int(), take(double(1..10), 28)))

def f1(val,value){
	return each(x->sum(x), val) + value
}

	val :: DOUBLE[] VECTOR
	def F1(val_){
		val = val_	
	}
	def getval(value){
		val = each(x->sum(x), val) + value
		return val	
	}		
}
val = arrayVector(take(2 3 5 2 5, 20).cumsum().int(), rand(100.0, 68))
f = F1(val)
f.getval(10.0)

@testing:case="test_topic_class_with_persional_function_DOUBLE[]"
def f1(val){
	return val * 0.56
}

class F1{
	val :: DOUBLE[] VECTOR
	def F1(val_){
		val = val_	
	}
	def getval(value){
		val = f1(value) +val
		return val	
	}		
}
val = arrayVector(take(2 3 5 2 5, 20).cumsum().int(), rand(100.0, 68))
f = F1(val)
assert 1, each(eqObj,f.getval(10.0), f1(10.0)+val)
assert 2, each(eqObj,f.getval(f.val), (f1(10.0)+val+f1(f1(10.0)+val)))

@testing:case="test_topic_class_with_presional_function_use_class_with_double[]"
class F1{
	name :: DOUBLE[] VECTOR
	def F1(name_){
		name = name_
	}
	def getS(){
		return array(DOUBLE[],0,0).append!([[1.0,2.0],[3.0,4.0]])
	}	
}

def speak(mutable F){
	str = F.getS()
	return str
}
f = F1(arrayVector(take(2 3 5 2 5, 20).cumsum().int(), rand(100.0, 68)))
assert 1, each(eqObj, speak(f) , array(DOUBLE[],0,0).append!([[1.0,2.0],[3.0,4.0]]))

@testing:case="test_topic_class_inhert_as_dict_value_INT[]"
class F {
	buySignalRSI :: INT[] VECTOR
	
	def F() {
		buySignalRSI = arrayVector(take(2 3 5 2 5, 20).cumsum().int(), take(double(11..20), 68))
	}
}
class Context : F {
	def Context() {
		F()
	}
}

d=dict(STRING,ANY)
c = Context()
d["c"] = c
assert 1,  each(eqObj,d["c"].buySignalRSI , arrayVector(take(2 3 5 2 5, 20).cumsum().int(), take(11..20, 68)))

}
assert 3, d.size() == 11
