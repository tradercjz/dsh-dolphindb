@testing:case="test_topic_class_with_scheduleJob"
if (isSingleMode == false) {
	classInit = "class F2{\n\tname :: STRING\n\tdef F2(){name = 'default'}\n\tdef getS(){return 'F2'}	\n\tdef setName(name_){\n\tname = name_\n\t}\n\tdef getName(){\n\t\treturn name\n\t}\n}"
	getTInit = "def getT(mutable F, newName){\n\tF.setName(newName)\n\tstr = F.getName()\n\tprint(str)\n}"
	speakInit = "def speak(mutable F, mutable d){\n\td['2'] = string(now())\n\tgetT(F, 'newName')\n\td['1'] = F.getName()\n\t}"
	
	clearAndRestartCluster(path,ctlSite,nodeSites,nodeAlias)
	waitCtlReady(ctlSite, "dfs://test_ready", "pt")
	for(nodeSite in nodeSites){
		waitChunkNodeInit(nodeSite)
	}
	waitChunkNodeOnline(ctlSite)
	conn = buildConn(nodeSites[0])
	remoteRun(conn, classInit)
	remoteRun(conn, getTInit )
	remoteRun(conn, speakInit )
	remoteRun(conn, "F = F2()")
	remoteRun(conn,"d = dict(STRING, ANY)")
	funcview = 'if("speak" in getFunctionViews()["name"]){\n\tdropFunctionView("speak")\n}'
	remoteRun(conn, funcview)
	remoteRun(conn, "addFunctionView(speak)")
	remoteRun(conn, 'scheduleJob("speak", "speakTest", speak{F, d}, temporalAdd(minute(now()), 1 6 11, "m"), date(now()), date(now())+1, "D")')
	sleep(30000)
	restartCluster(path, ctlSite, nodeSites, nodeAlias)
	waitCtlReady(ctlSite, "dfs://test_ready", "pt")
	for(nodeSite in nodeSites){
		waitChunkNodeInit(nodeSite)
	}
	waitChunkNodeOnline(ctlSite)
	
	sleep(60000)
	conn = buildConn(nodeSites[0])
	assert 1, "speak" in conn(getFunctionViews)["name"]
	assert 2, "speakTest" in conn(getRecentJobs)["jobDesc"]
	conn(dropFunctionView{"speak"})
	assert 3, not "speak" in conn(getFunctionViews)["name"]
}
