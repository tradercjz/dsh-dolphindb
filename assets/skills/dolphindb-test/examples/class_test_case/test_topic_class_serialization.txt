@testing:case="test_topic_class_with_persional_function"
def f1(val){
	return val * 0.56
}

class F1{
	val :: DOUBLE
	def F1(){}
	def getval(value){
		val = value + f1(value)
		return val	
	}		
}

f = F1()
assert 1, each(eqObj,f.getval(10.0), (10+f1(10)))
assert 2, each(eqObj,f.val , (10+f1(10)))
assert 3, each(eqObj,f.getval(f.val), (15.6+f1(15.6)))

@testing:case="test_topic_class_with_presional_function_2"
def f1(val){
	return val * 0.56
}

class F1{
	val :: DOUBLE
	def F1(){}
	def getval(value){
		val = value + f1(val)
		return val	
	}		
}
class F2:F1{
	def F2(){
		val = 0.0	
	}
	def getval2(beta){
		return getval(beta)\beta
	}
}

f = F2()
assert 1, each(eqObj,f.getval(10.0), (10+f1(0)))
assert 2, each(eqObj,f.val , (10+f1(0)))
assert 3, each(eqObj,f.getval2(f.val), (10+f1(10))\10)
