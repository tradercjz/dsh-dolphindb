class A{
	name::STRING
	def A(name_){
		name = name_
	}	
	def sql1(dbName, tableName, colName){
		whereCondition = parseExpr(colName+"=`A1")
		script = sql(select=sqlColAlias(<avg(value)>), from=loadTable(dbName, tableName), where=whereCondition, groupBy=sqlCol(name))
		return eval(script)
	}

class demo{
	name :: ANY
	t :: ANY
	def demo(name_) {
		name = name_
		t = table(take(1..3, 2000) as id, rand(`a1`a2`a3, 2000) as colName, rand(100.0, 2000) as value)
	}
	def whereCol() {
		t_pre = select * from t where colName = name
		return t_pre
	}
	def groupbyCol(func){
		t_pre = select func(value) from t   group by name[colName]	
		return t_pre
	}
	def whereCol2(name2){
		t_pre = select value > name from t where colName = name2
		return t_pre
	}	
}
new = demo("a1")
res = new.whereCol()
ext  = select * from new.t where colName = "a1"
assert 2, each(eqObj, res.values(), ext.values())


d1 = dict(`a1`a2`a3, 1 1 2)

res.sortBy!(res.colNames())
ext.sortBy!(ext.colNames())
assert 3, each(eqObj, res.values(), ext.values())
