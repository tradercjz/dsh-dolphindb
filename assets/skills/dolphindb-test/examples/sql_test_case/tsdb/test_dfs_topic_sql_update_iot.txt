def createTables(){
	if(existsDatabase("dfs://test_iot_update")){
		dropDatabase("dfs://test_iot_update")
	}
	db1=database("",VALUE,2021.03.01+0..30)
	db2=database("",HASH,[INT,5])
	db=database("dfs://test_iot_update",COMPO,[db1,db2],,"TSDB")
	n=6000
	t = table(take(2021.03.01 +1..30,n) as ts, take(time(10) +1..30,n) as time, take(1..300000,n) as qty , take(`A`B`C`D,n) as sym,take(10.5+1..100,n) as price)
	dt=db.createPartitionedTable(t,`dt,`ts`qty, {ts:"delta", qty:"delta"},`qty`time)
	empt=db.createPartitionedTable(t,`empt,`ts`qty,{ts:"delta", qty:"delta"},`qty`time)
	wt=db.createTable(t,`wt,,`qty`time)
	dt.append!(t)
	wt.append!(t)
	n=6
	t1 = table(take(2021.03.01 +1..6,n) as ts,take(time(10) +1..6,n) as time, take(1..10000,n) as qty , take(`A`B``D,n) as sym,rand(100.0,n) as price)
	dt1=db.createPartitionedTable(t,`dt1,`ts`qty, {ts:"delta", qty:"delta"},`qty`time)
	wt1=db.createTable(t,`wt1,,`qty`time)
	dt1.append!(t1)
	wt1.append!(t1)

}
