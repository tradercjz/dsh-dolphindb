def createTsdbDB(dbName){
	if(existsDatabase(dbName)){
		dropDatabase(dbName)
	}
	db=database(dbName, VALUE, 2012.01.01..2012.01.30,  ,"TSDB")
}

def createTsdbPartTable1(dbName, tableName){
	db=database(dbName)
	t=table(1:0, `date`time`sym`price`val, [DATE, SECOND, SYMBOL, INT, DOUBLE])
	db.createPartitionedTable(t, tableName, "date", , ["date"],sortKeyMappingFunction=[hashBucket{,10}])
}

@testing:case="test_dfs_topic_sql_update_where_one_sortkey"
dbName="dfs://test_dfs_topic_sql_update_where"
tableName=`pt1
tableName1=`dt1
createTsdbDB(dbName)
createTsdbPartTable1(dbName, tableName)
createTsdbTable1(dbName, tableName1)
n=50000
date=take(2012.01.01..2012.01.30,n).sort()
time=second(now())+1..n
sym=rand("v"+string(1..40),n)
price=rand(20,n)
val=rand(20.0,n)
t=table(1:0, `date`time`sym`price`val, [DATE, SECOND, SYMBOL, INT, DOUBLE])
insert into t values(date,time,sym,price,val)
pt=loadTable(dbName,tableName)
pt.append!(t)
dt=loadTable(dbName,tableName1)
dt.append!(t)
update pt set val=0 where date>2012.01.20
update dt set val=0 where date>2012.01.20
update t set val=0 where date>2012.01.20
resp=select * from loadTable(dbName,tableName) order by date,time,sym,price,val
resd=select * from loadTable(dbName,tableName1) order by date,time,sym,price,val
ex=select * from t order by date,time,sym,price,val
assert 1,each(eqObj,resp.values(),ex.values())
assert 2,each(eqObj,resd.values(),ex.values())

update pt set val=1 where date==2012.01.10
update dt set val=1 where date==2012.01.10
update t set val=1 where date==2012.01.10
resp=select * from loadTable(dbName,tableName) order by date,time,sym,price,val
resd=select * from loadTable(dbName,tableName1) order by date,time,sym,price,val
ex=select * from t order by date,time,sym,price,val
assert 3,each(eqObj,resp.values(),ex.values())
