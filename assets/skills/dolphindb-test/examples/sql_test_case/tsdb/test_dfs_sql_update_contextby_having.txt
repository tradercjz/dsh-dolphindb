def nodeCompaction(dbName){
	suffix=substr(dbName, 5)
	chunkIds=exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
	for(x in chunkIds){
		  triggerIotCompaction(x)
	}
}

dbName = "dfs://update_context_by_having"
tableNames=("pt"+string(1..6)) join ("dt"+string(1..6))
for(tableName in tableNames){
	pt=loadTable(dbName,tableName)
	t=select * from pt
	//empty filter
	update pt set price1 = price1+price2 context by date(timestampv),ID having count(price2)<-2
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 1,each(eqObj,re.values(),ex.values())
	//all filter
	t=select * from pt
	update pt set price1 = price1+cummax(price2) context by date(timestampv),ID having count(price2)>-10
	update t set price1 = price1+cummax(price2) context by date(timestampv),ID 
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 2,each(eqObj,re.values(),ex.values())
	//part group filter 
	t=select * from pt
	update pt set price2 = cummax(price2)-price2 context by date(timestampv),ID having avg(price2)>50
	update t set price2 = cummax(price2)-price2 context by date(timestampv),ID having avg(price2)>50
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 3,each(eqObj,re.values(),ex.values())
	//part filter in group
	t=select * from pt
	update pt set price1 = cummin(price2) context by date(timestampv),ID having avg(price2)>price1
	update t set price1 = cummin(price2) context by date(timestampv),ID having avg(price2)>price1
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 4,each(eqObj,re.values(),ex.values())
	//part group part filter 
	t=select * from pt
	update pt set price2 = mstd(price1,2) context by date(timestampv) having avg(price2)>price1 and count(*)>500
	update t set price2 = mstd(price1,2)  context by date(timestampv) having avg(price2)>price1 and count(*)>500
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 5,each(eqObj,re.values(),ex.values())	
}

@testing:case="test_dfs_sql_update_context_by_having_fucntions"
defg f1(x){
	return avg(x)+1
}
defg f2(x,y){
	return mavg(x,2)+y
}
dbName = "dfs://update_context_by_having"
tableNames=("pt"+string(1..6)) join ("dt"+string(1..6))
for(tableName in tableNames){
	pt=loadTable(dbName,tableName)
	//def
	t=select * from pt
	update pt set price1 = price1+1 context by date(timestampv),symbolv having f1(ID)<=50
	update t set price1 = price1+1 context by date(timestampv),symbolv having f1(ID)<=50
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 1,each(eqObj,re.values(),ex.values())
	
	t=select * from pt
	update pt set price1 = price1+1 context by symbolv,date(timestampv)having f2(ID,price2)<=50
	update t set price1 = price1+1 context by symbolv,date(timestampv) having f2(ID,price2)<=50
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 2,each(eqObj,re.values(),ex.values())
	//Aggregate 
	t=select * from pt
	update pt set symbolv =symbolv+"3",datev=date(timestampv)+1 context by date(timestampv) having beta(price1,price2+1)>0 and std(sort)<=28000
	update t set symbolv =symbolv+"3",datev=date(timestampv)+1 context by date(timestampv) having beta(price1,price2+1)>0 and std(sort)<=28000
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 3,each(eqObj,re.values(),ex.values())
	//no Aggregate
	t=select * from pt
	update pt set symbolv =symbolv+"3",datev=date(timestampv)+1 context by date(timestampv),symbolv csort ID having denseRank(ID)==0
	update t set symbolv =symbolv+"3",datev=date(timestampv)+1 context by date(timestampv),symbolv csort ID having denseRank(ID)==0
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 4,each(eqObj,re.values(),ex.values())
	//no Aggregate && Aggregate
	t=select * from pt
	update pt set symbolv =symbolv+"3",datev=date(timestampv)+1 context by symbolv ,date(timestampv)csort ID having denseRank(ID)>=10 and (f1(ID)<=50 or beta(price1,price2+1)>0)
	update t set symbolv =symbolv+"3",datev=date(timestampv)+1 context by symbolv ,date(timestampv)  csort ID having denseRank(ID)>=10 and (f1(ID)<=50 or beta(price1,price2+1)>0)
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 5,each(eqObj,re.values(),ex.values())
	//expression
	t=select * from pt
	update pt set symbolv =symbolv+"3",datev=date(timestampv)+1 context by ID,symbolv  having price1-price2>3 	
	update t set symbolv =symbolv+"3",datev=date(timestampv)+1 context by ID,symbolv  having price1-price2>3 
	re=select * from pt order by sort
	ex=select * from t order by sort
	assert 6,each(eqObj,re.values(),ex.values())
}
