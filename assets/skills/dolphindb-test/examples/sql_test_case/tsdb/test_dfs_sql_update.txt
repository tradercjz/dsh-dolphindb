// 创建数据库
def createSingleModeFactorTB(dbname, tbname, initialdb=false, initialtb=false){
    // initialdb=true 会删库
    if(initialdb==true){
        try{dropDatabase(dbname)}catch(ex){print(ex)}
    }

    if(existsDatabase(dbname)){     // 如果有同名数据库，直接使用已有的数据库
        db = database(dbname)
    }else{                          // 如果没有同名数据库，创建一个
        db1 = database(, VALUE, 2020.01M + 3 * (0..20)) // 分区的第一个规则采用股票月份分区，这里的2020.01M..2020.12M只是一个案例，未来新的数据会按照这个案例插入tb
        db2 = database(, VALUE, [`demo])   //用因子symbol分区，其目的是构造因子窄表，便于后续的增删因子，以及加速因子查询
        db = database(dbname, COMPO, [db1, db2], , "TSDB", atomic='CHUNK') // 采用TSDB记录数据，分区规则按TradeTime按月分区 + factorName按值分区的方式
    }

    // intialtb=true 会删表
    if(initialtb==true){
        try{dropTable(db, tbname)}catch(ex){print(ex)}
    }

    if(existsTable(dbname, tbname)){    // 如果有同名表，直接使用已有的表
        pt = loadTable(dbname, tbname)
    }else{
        colname = ['time','securityCode','factorName','factorValue']
        coltype = ['TIMESTAMP','SYMBOL','SYMBOL','DOUBLE']
        schematb = table(1:0, colname, coltype)
        pt = db.createPartitionedTable(schematb, tbname, partitionColumns=`time`factorName, sortcolumns=`securityCode`time, sortKeyMappingFunction=[hashBucket{,500}])
    }
    return pt
}

dbname, tbname = "dfs://test", "test"
createSingleModeFactorTB(dbname, tbname, initialdb=true, initialtb=false)
