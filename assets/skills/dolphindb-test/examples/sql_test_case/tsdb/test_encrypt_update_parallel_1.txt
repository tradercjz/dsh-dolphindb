@testing:case="test_topic_create_pointManager_prepare"
def pnodeRunAllNodes(function, nodes=NULL, addNodeAlias=true){
	allNodes = exec name from rpc(getControllerAlias(),getClusterPerf) where mode=0 or mode=4
	if((isNull(nodes)).any() && addNodeAlias==true){
		return pnodeRun(function,nodes=allNodes, addNodeAlias=true)
	}else if((isNull(nodes)).any() && addNodeAlias==false){
		return pnodeRun(function,nodes=allNodes,addNodeAlias=false)
	}else if((!isNull(nodes)).any() && addNodeAlias==true){
		return pnodeRun(function,nodes=nodes, addNodeAlias=true)
	}else if((!isNull(nodes)).any() && addNodeAlias==false){
		return pnodeRun(function,nodes=nodes, addNodeAlias=false)
	}
}

def nodeCompaction(dbName,level){
	 suffix = substr(dbName, 5)
	 chunkIds = exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
	 for(x in chunkIds){
	 	triggerTSDBCompaction(x,level)
	 }
}


def getAllChunksMeta(){
	return pnodeRun(getTabletsMeta)
}

def checkChunkByReadTabletChunk(dbName, targetTableName, targetChunk){
	for(i in 1..10){
		dbName_len=substr(dbName, 5).strlen()
		chunkInfo=exec *, substr(dfsPath, dbName_len) as chunkPath from pnodeRun(getTabletsMeta) where chunkId=targetChunk, tableName=targetTableName
		length=exec count(*) from loadTable(dbName, targetTableName)
		re=false
		try{
			res1=rpc(chunkInfo[`node][0], readTabletChunk, string(targetChunk), dbName, chunkInfo[`dfsPath][0], targetTableName, 0, length)
			res2=rpc(chunkInfo[`node][1], readTabletChunk, string(targetChunk), dbName, chunkInfo[`dfsPath][1], targetTableName, 0, length)
			re=each(eqObj, res1.values(), res2.values()).all()
			if(re==true){
				break
			}
		}catch(ex){
			sleep(1000)
			if(re==false  && i==10){
				throw dbName+"/"+targetTableName+" "+string(targetChunk)+" different"
			}
		}
	}
}

def getData(dates,names, datatype, n, dbName, tableName){
	ta = table(n:n, `id`name`flag`createTime, [STRING, STRING,STRING,TIMESTAMP])
	ta[`id] = take(string(datatype), n)
	ta[`name] = take(names, n)
	ta[`flag] =  take(symbol(rand(string(1..100000),400)+".SZ"), n)
	ta[`createTime] =  concatDateTime(rand(dates, n), time(rand(1000000..1040000, n)))
	ta[`value] = rand(`AA`BB`CC`DD`EE`FF`GG`HH, n)
	tmp1 = select count(*) from loadTable(dbName, tableName) group by name
	delete  ta from ej(ta, tmp1, `name)
	return ta
}

def getValueData(counts, datatype){
	if(string(datatype) in ["symbol", "string"]){
		value = datatype(take(`AA`BB`CC`DD`EE`FF`GG`HH, counts))
	}else{
		value = datatype(rand(100.0, counts).decimal128(3))
	}	
	return value
}

def cancelJobs(){
	res = select node , jobId from pnodeRun(getRecentJobs) where endTime is null and like(jobDesc,"%query%")
	for(index in 0:res.rows()){
		rpc(res.row(index).node, cancelJob, res.row(index).jobId)	
	}
}

def updateData1(dbName, tableName, d, names, data,  mutable exData){
	try{
		update loadTable(dbName, tableName) set value = data where name  = names
		exTable = exData[d]
		//update exTable set value = data where  name  = names
		tmp1 = select * from exTable where name  = names
		delete from exTable where name  = names
		replaceColumn!(tmp1, `value, take(data, tmp1.rows()))
		exTable.append!(tmp1)
	}catch(ex){
		print(exec count(*) from exTable where  name  = names)	
		print(data.size())
	}
	exData[d] = exTable
}

def updateData2(dbName, tableName, d,  mutable exData){
	for(i in 0:10){
		tmp =exec last(id) as id, count(*) from loadTable(dbName, tableName) where date(createTime) = d group by name

		counts = (exec count from tmp where name = names)[0]
		//datatype = rand([char, short, int, long, float, double, string, symbol], 1)[0]
		dtype = (exec id from tmp where name = names)[0]
		datatype  = funcByName(iif(dtype=="symbol", "string", dtype))
		data = getValueData(counts, datatype)[0]
		updateData1(dbName, tableName, d, names, data, exData)
	}
}

def getData(dates,names, n){
	n1 = n/8
	n2 = n - (n1*7)
	datatypes = [string, symbol,char, short, int, long, float, double]
	tmpTable =  table([string(),string()] as id, `A`A as name, `A`C as flag, timestamp(0 0) as createTime,["1", "a"] as value)
	start = 0
	end = 0
	for(i in 0:datatypes.rows()){
		num = iif(i == 6, n1, n2)
		datatype = datatypes[i]
		val = rand(`AA`BB`CC`DD`EE`FF`GG`HH, num)
		ta = table(num:num, `id`name`flag`createTime, [STRING, STRING,STRING,TIMESTAMP])
		ta[`id] = take(string(datatype), num)
		ta[`name] = take(names, num)
		start = end + 1
		end = start + 999
		ta[`flag] =  take(rand(string(start..end),10)+".SZ", num)
		ta[`createTime] =  concatDateTime(rand(dates, num), time(rand(1000000..1040000, num)))
		ta[`value] = val
		tmpTable.append!(ta)
	}
	return select * from tmpTable where id is not null
}
