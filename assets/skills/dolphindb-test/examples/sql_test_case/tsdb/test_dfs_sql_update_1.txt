@testing:case = "test_dfs_sql_update_createTables_prepare"
def createTables(){
	if(existsDatabase("dfs://test_update")){
		dropDatabase("dfs://test_update")
	}
	db1 = database("", VALUE, 2021.03.01+0..10)
	db2 = database("", HASH, [INT, 5])
	db = database("dfs://test_update", COMPO, [db1, db2], , "TSDB")
	n = 6000
	t = table(take(2021.03.01 + 1..30, n) as ts, take(time(10) + 1..30, n) as time, take(1..300000, n) as qty, take(`A`B`C`D, n) as sym, rand(10.0, n) as price)
	dt = db.createPartitionedTable(t, `dt, `ts`qty, {ts: "delta", qty: "delta"}, `qty`ts)
	empt = db.createPartitionedTable(t, `empt, `ts`qty, , `qty`ts)
	wt = db.createTable(t, `wt, , `qty`ts)
	dt.append!(t)
	wt.append!(t)
	n = 6
	t1 = table(take(2021.03.01 + 1..6, n) as ts, take(time(10) + 1..6, n) as time, take(1..10000, n) as qty, take(`A`B`C`D, n) as sym, rand(100.0, n) as price)
	dt1 = db.createPartitionedTable(t, `dt1, `ts`qty, , `qty`ts).append!(t1)
	wt1 = db.createTable(t, `wt1, , `qty`ts).append!(t1)
}
