def nodeCompaction(dbName){
	 suffix = substr(dbName, 5)
	 chunkIds = exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
	 for(x in chunkIds){

	 }
}

def waitChunkCompaction(dbName){
	allChunks = exec chunkId from rpc(getControllerAlias(),getClusterChunksStatus) where like(file,"%/"+substr(dbName,6)+"/%")
	do{
		unable_compaction = exec count(*) from pnodeRunAllNodes(getTSDBCompactionTaskStatus) where isNull(endTime) and chunkId in allChunks
		if(unable_compaction==0){
			print("All Chunks Finish")
			break	
		}else{
			print(unable_compaction+ " chunks are waiting")
			sleep(1000)	
		}
	}while(true)
	errorMessageNum = exec count(*) from pnodeRunAllNodes(getTSDBCompactionTaskStatus) where chunkId in allChunks and isNull(errorMessage)==false
	if(errorMessageNum!=0){
		throw "some chunks merge fail,please check"
	}else{
		print "merge successful"
	}
}
def checkMerge(dbName,tableName,exLevel){
	waitChunkCompaction(dbName)
	allChunks = exec chunkId from pnodeRunAllNodes(getAllChunks) where like(dfsPath,"%/"+substr(dbName, 6, dbName.strlen())+"/%") and like(dfsPath,"%.tbl")==false and like(dfsPath,"%domain")==false
	chunkNum = ((exec distinct(chunkId) from pnodeRunAllNodes(getTSDBMetaData) where like(chunkPath,"%/"+substr(dbName, 6, dbName.strlen())+"/%") and like(table,tableName+"%")).size())*2
	for(i in 1..20){
	 	levelFileNum = exec count(*) from pnodeRunAllNodes(getTSDBMetaData) where like(chunkPath,"%/"+substr(dbName, 6, dbName.strlen())+"/%") and like(table,tableName+"%") and level							=exLevel
		 if(levelFileNum == chunkNum){
	 		print("levelFilesNum is expected")
	 		break
		 }else{
	 		if(i == 20){
	 			throw "levelFilesNum is not expected after " +string(i)+ "seconds"
	 		}
	 		print("levelFilesNum is not expected,wiat one second")
	 		sleep(1000)
		 }
	}
	ex1 = take(string(exLevel),chunkNum)
	for(i in 1..60){
		levels = exec left(files,1) from pnodeRunAllNodes(getTSDBMetaData) where like(chunkPath,"%/"+substr(dbName, 6, dbName.strlen())+"/%") and like(table,tableName+"%") and level=exLevel
		if((levels==ex1).all()){
			print "merge to "+exLevel+" level successful"
			break
		}else{
			if(i == 60){
				throw "merge to "+exLevel+" level fail after "+string(i)+" seconds"
			}
			sleep(1000)
		}
	}
}

def createDB(dbName,  partitionType, partitionSchemeList) {
	if (existsDatabase(dbName)){
       		 dropDatabase(dbName)
  	  }
	db = database(dbName, partitionType, partitionSchemeList, engine='TSDB')
	return db
}

def createDB2(dbName, partitionType1,partitionType2, partitionSchemeList1, partitionSchemeList2) {
	if (existsDatabase(dbName)){
       		 dropDatabase(dbName)
  	  }
	db1 = database("", partitionType1, partitionSchemeList1)
	db2 = database("", partitionType2, partitionSchemeList2)
	db = database(dbName, COMPO, [db1,db2], engine='TSDB')
	return db
}
