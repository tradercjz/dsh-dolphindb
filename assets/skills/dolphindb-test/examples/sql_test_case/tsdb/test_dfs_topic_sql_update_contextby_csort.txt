@testing:case="test_topic_sql_contextBy_csort_prepare"
def create_pt_dt(t, partiCol){
    dbName = "dfs://test_update_contextBy_csort"
    if(existsDatabase(dbName)){
        dropDatabase(dbName)
    }
    db = database(dbName, VALUE, 1..10, ,"TSDB")
    pt = createPartitionedTable(db, t, `pt, partiCol, sortColumns=`id).append!(t)
    dt = createTable(db, t, `dt, sortColumns=`id).append!(t)
}

def nodeCompaction(dbName){
	suffix=substr(dbName, 5)
	chunkIds=exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
	for(x in chunkIds){
		  triggerIotCompaction(x)
	}
}
