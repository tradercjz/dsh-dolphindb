def clearAndCreateDb(dbName){
    if(existsDatabase(dbName)){
		dropDatabase(dbName)	
	}
	db = database(dbName, VALUE, 1 2 3,,"TSDB")
    return db
}

def generatePartitionedTable(db, dbName, tableName, rows){
	t = create_t(rows)
	pt = createPartitionedTable(db, t, tableName,  `id, sortColumns=`id).append!(t)
	return pt
}

def generateDimensionTable(db, dbName, tableName, rows){
	t = create_t(rows)
	dt = createTable(db, t, tableName, sortColumns=`id).append!(t)
	return dt		
}

def generatePartitionedTable_sort(db, dbName, tableName, rows){
	t = create_sort_t(rows)
	pt = createPartitionedTable(db, t, tableName,  `id, sortColumns=`id).append!(t)
	return pt
}

def generateDimensionTable_sort(db, dbName, tableName, rows){
	t = create_sort_t(rows)
	dt = createTable(db, t, tableName, sortColumns=`id).append!(t)
	return dt		
}


def generatePartitionedTable_use_otherTable(db, dbName, tableName, t){
	pt = createPartitionedTable(db, t, tableName,  `id, sortColumns=`id).append!(t)
	return pt
}

def generateDimensionTable_use_otherTable(db, dbName, tableName, t){
	dt = createTable(db, t, tableName, sortColumns=`id).append!(t)
	return dt		
}
