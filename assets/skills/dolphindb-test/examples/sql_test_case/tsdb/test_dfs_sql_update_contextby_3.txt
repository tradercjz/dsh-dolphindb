def createDB(partitionType, partitionScheme) {
	if (existsDatabase("dfs://test")) {
	    dropDatabase("dfs://test")
	}
	db = database("dfs://test", partitionType, partitionScheme, , "TSDB")

	return db
}

def createPTable(n, db, tName, pcol, pcolStr, col, colStr) {
	id = rand(n/2, n)
	t = table(id as id, pcol as pcol, col[0][1] as col0, col[1][1] as col1, col[2][1] as col2, col[3][1] as col3, col[4][1] as col4, col[5][1] as col5, col[6][1] as col6, col[7][1] as col7, col[8][1] as col8, col[9][1] as col9, col[10][1] as col10, col[11][1] as col11, col[12][1] as col12, col[13][1] as col13, col[14][1] as col14, col[15][1] as col15, col[16][1] as col16, col[17][1] as col17, col[18][1] as col18, col[19][1] as col19, col[20][1] as col20, col[21][1] as col21, col[22][1] as col22, col[23][1] as col23, col[24][1] as col24, col[25][1] as col25, col[26][1] as col26, col[27][1] as col27, col[3][1] as scol0, col[5][1] as scol1)
	// multi sortcolumn
	pt = db.createPartitionedTable(t, tName, `pcol, , `id`scol0`scol1, sortKeyMappingFunction=[hashBucket{,5}, def (x){return x+1}])
	loadTable("dfs://test", tName).append!(t)
	pnodeRunAllNodes(flushTSDBCache)
	
	pt = loadTable("dfs://test", tName)

	return pt
}
