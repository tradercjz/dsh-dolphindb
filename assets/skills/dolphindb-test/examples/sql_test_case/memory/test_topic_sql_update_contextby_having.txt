mytype=int
def testUpdateContextBy(n,close,high,mytype){
	closev=mytype(close)
	highv=mytype(high)
	t=table(rand([1, 0], n) as cb, symbol(rand("00000"+string(1..9)+".SZ", n)) as symbol, closev as close, highv as high)
	s=exec sum(close) from t context by symbol 
	tmp1=select sum(cb) as cb,symbol,min(close) as close,max(high) as high from t context by symbol having  sum(close)>s[0]
	tmp2=select * from t context by symbol having  sum(close)<=s[0]
	ex=select * from unionAll(tmp1,tmp2) order by cb,symbol,close,high
	update t set cb = sum(cb),close=min(close),high=max(high) context by symbol having  sum(close)>s[0]
	res=select * from t order by cb,symbol,close,high
	return each(eqObj,res.values(),ex.values())
}

def testUpdateContextBy(n,close,high,mytype){
	closev=mytype(close)
	highv=mytype(high)
	t=table(rand([1, 0,], n) as cb, symbol(rand("00000"+string(1..9)+".SZ", n)) as symbol, 1..n as flag, closev as close, highv as high)
	data=exec avg(close) as a,med(high) as m,abs(close-high) as abs from t context by symbol 

	tmp1=select count(cb) as cb,symbol,flag,last(close) as close,max(high) as high from t context by symbol having  avg(close)>=data[`a][0] and med(high)=data[`m][0]
	tmp2=select * from t context by symbol having  avg(close)<data[`a][0] or med(high)!=data[`m][0]
	ex=select cb,symbol,flag,close,high from unionAll(tmp1,tmp2) order by flag
	replaceColumn!(ex,`high,mytype(ex.high))
	update t set cb = count(cb),close=last(close),high=max(high) context by symbol having  avg(close)>=data[`a][0] and med(high)=data[`m][0]
	res=select * from t order by flag
	result1=each(eqObj,res.values(),ex.values())
	
	tmp1=select count(cb) as cb,symbol,flag,last(close) as close,max(high) as high from t context by symbol having  avg(close)<= med(high)
	tmp2=select * from t context by symbol having  avg(close)> med(high)
	ex=select cb,symbol,flag,close,high from unionAll(tmp1,tmp2) order by flag
	replaceColumn!(ex,`high,mytype(ex.high))
	update t set cb = count(cb),close=last(close),high=max(high) context by symbol having  avg(close)<= med(high)
	res=select * from t order by flag
	result2=each(eqObj,res.values(),ex.values())
	
	tmp1=select count(cb) as cb,symbol,flag,last(close) as close,max(high) as high from t context by symbol having  abs(close-high) >=data[`abs][0]
	tmp2=select * from t context by symbol having  abs(close-high)<data[`abs][0]
	ex=select cb,symbol,flag,close,high from unionAll(tmp1,tmp2) order by flag
	replaceColumn!(ex,`high,mytype(ex.high))
	update t set cb = count(cb),close=last(close),high=max(high) context by symbol having  abs(close-high) >=data[`abs][0]
	res=select * from t order by flag
	result3=each(eqObj,res.values(),ex.values())
	return result1&&result2&&result3
}
