def arrayvector_append(v,datatype,n,k){
	av = array(datatype, 0, n)
	for(i in 0:n){
		tempv = rand(v,k)
		av.append!([tempv])
	}
	return av
}

def arrayvector_append_noteq(v,n){
	zeroNum = (n/10)
	kt = shuffle(take([1,2,10],n-6).join(take([256,65535,65536],6)))
	av = arrayVector(int(cumsum(kt)),rand(v,cumsum(kt)[n-1])).append!(take([[]],zeroNum))
	return rand(av,n)
}
