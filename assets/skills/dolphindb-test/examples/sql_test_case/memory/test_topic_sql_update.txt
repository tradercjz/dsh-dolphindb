@testing:case="test_function_decimal_update_ej"
def f1(sca){
	x1 = decimal32([0.78,0.46,0.51,0.01],sca)
	t1= table(1 2 3 3 as id, x1 as value)
	t2 = table(5 3 1 as id, [0.3, 0.5, 0.8] as qty)

	}
