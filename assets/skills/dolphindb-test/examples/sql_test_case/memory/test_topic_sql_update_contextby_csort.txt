datatypes=["char", "int", "short", "long", "float", "double", "date", "month", "time", "minute", "second", "datetime", "timestamp", "nanotime", "nanotimestamp","string","symbol","bool", "decimal32", "decimal64", "decimal128"]
def checkContextByCsort(n, dataType){
    id = take(1..3, n)
    sym = take(symbol("sym"+string(1..3)), n)
    val = rand(10.0, n)
    if(dataType=="symbol"){
        price = funcByName(dataType)(string(take(1..1000000 ,n)))  
    }else if(dataType in ["decimal32", "decimal64", "decimal128"]){
        price = funcByName(dataType)(take(1..1000000 ,n), 2)
    }else{
        price = funcByName(dataType)(take(1..1000000 ,n))
    }
    t = table(id, sym, val, price)
    tmp1 = select id, sym, msum(val,2), price from (select * from t where id=1 order by price asc)
    tmp2 = select id, sym, msum(val,2), price from (select * from t where id=2 order by price asc)
    tmp3 = select id, sym, msum(val,2), price from (select * from t where id=3 order by price asc)
    update t set val=msum(val,2) context by id csort price asc
    ex = select * from (select * from tmp1 union all select * from tmp2 union all select * from tmp3)
    ans = select * from t
    if(ex.rows()){ex.sortBy!(ex.columnNames())}
    if(ans.rows()){ans.sortBy!(ans.columnNames())}
    result1 = each(eqObj, ex.values(), ans.values())

    t = table(id, sym, val, price)
    tmp1 = select id, sym, msum(val,2), price from (select * from t where id=1 order by price desc)
    tmp2 = select id, sym, msum(val,2), price from (select * from t where id=2 order by price desc)

def checkContextByCsort(n, dataType){
    id = take(1..3, n)
    sym = take(symbol("sym"+string(1..3)), n)
    val = rand(10.0, n)
    if(dataType=="symbol"){
        price = funcByName(dataType)(string(take(1..1000000 ,n)))  
    }else if(dataType in ["decimal32", "decimal64", "decimal128"]){
        price = funcByName(dataType)(take(1..1000000 ,n), 2)
    }else{
        price = funcByName(dataType)(take(1..1000000 ,n))
    }
    t = table(id, sym, val, price)
    tmp1 = select id, sym, msum(val,2), price from (select * from t where id=1 order by price asc, sym desc, val asc)
    tmp2 = select id, sym, msum(val,2), price from (select * from t where id=2 order by price asc, sym desc, val asc)
    tmp3 = select id, sym, msum(val,2), price from (select * from t where id=3 order by price asc, sym desc, val asc)
    update t set val=msum(val,2) context by id csort price asc, sym desc, val asc
    ex = select * from (select * from tmp1 union all select * from tmp2 union all select * from tmp3)
    ans = select * from t
    if(ex.rows()){ex.sortBy!(ex.columnNames())}
    if(ans.rows()){ans.sortBy!(ans.columnNames())}
    result1 = each(eqObj, ex.values(), ans.values())

    return result1
}
