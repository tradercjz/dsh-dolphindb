#include "checkChunk.txt"

def myTriggerPKEYCompaction(dbName){
	flushPKEYCache()
	suffix=substr(dbName, 5)
	chunkIds=exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
    triggerPKEYCompaction(chunkIds, false)
}

def myUpdatePKEYDeleteBitmap(dbName){
	suffix = substr(dbName, 5)
	chunkIds = exec chunkId from getAllChunks() where like(dfsPath, suffix+"/%"), type==1, flag==0
	updatePKEYDeleteBitmap(chunkIds)
}
