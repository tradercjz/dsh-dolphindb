
@testing:case="test_dfs_topic_delete_where_single_primaryKey_int"
login(`admin,`123456)
dbName = "dfs://delete_where"
if(existsDatabase(dbName)){
	dropDatabase(dbName)
}
db = database(directory=dbName, partitionType=HASH, partitionScheme=[SYMBOL,50], engine="PKEY")
n = 100000
t = table(n:n, [`intv,`doublev,`symv,`timestampv,`datev,`blobv], [INT, DOUBLE, SYMBOL, TIMESTAMP,DATE,BLOB])
t[`intv] = rand(100,n)
t[`doublev] = rand(100.0,n)
t[`symv] = rand("A"+string(1..100),n)
t[`timestampv] = temporalAdd(2012.06.13T13:30:10.000,1000*( take(1..1000,n).sort!() ),"s")
t[`datev] =  rand(2020.01.01..2020.01.10 ,n)
t[`blobv] = rand("abcdefg"+string(1..100),n)
x1 = [<intv>10>, <intv<10>, <intv=10>, <intv!=10>, <intv in 1..50>, <not intv in 1..50>, <intv between 1:50>]
t = select * from t where !isDuplicated([symv,intv],LAST)
for(ac in x1){
	tem = t
	pt = database(dbName).createPartitionedTable(table=t, tableName=`pt1, partitionColumns=`symv, primaryKey=`symv`intv).append!(tem)
	print(sqlDelete(pt, where=ac))
	sqlDelete(pt, where=ac).eval()
	sqlDelete(tem, where=ac).eval()
	re = select * from pt order by intv,doublev,symv,timestampv,datev,blobv
	ex = select * from tem order by intv,doublev,symv,timestampv,datev,blobv
	assert 1,each(eqObj,re.values(),ex.values())
	dropTable(database(dbName),`pt1)	
}
//where primaryKey value not exist
x1 = [<intv>100>, <intv<-1>, <intv=200>, <intv in 200..300>, <not intv in 0..100>, <intv between 200:300>]
pt = database(dbName).createPartitionedTable(table=t, tableName=`pt1, partitionColumns=`symv, primaryKey=`symv`intv).append!(t)
for(ac in x1){
	print(sqlDelete(pt, where=ac))
	sqlDelete(pt, where=ac).eval()
	re = exec count(*) from pt 
	assert 2,eqObj(re,size(t))
}
