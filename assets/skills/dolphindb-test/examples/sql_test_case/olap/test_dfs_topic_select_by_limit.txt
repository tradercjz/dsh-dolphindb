def create_table(bdate, n){
    symbol_data = take(`symbol_data, n)
    string_data = take(`string_date, n)
    date_data = bdate+(1..n)
    double_data = take(1..n, n)

    OID = string(take(1..n, n))     
    ENTRY_ID = `HG + string(take(1..n, n))      
    ENTRY_ITEM_NO = ENTRY_ID+`LH + string(take(1..n, n))  
    DECLARE_DATE = date_data   
    CUSTOM_MASTER = symbol_data 
    DECLARE_CODE = symbol_data   
    DECLARE_NAME = symbol_data    
    OWNER_CODE = symbol_data      
    OWNER_NAME = symbol_data      
    TRADE_CODE  =  symbol(string(take(1..n, n)))  
    TRADE_NAME = symbol_data      
    DISTINATE_PORT = symbol_data  
    DISTRICT_CODE = symbol_data  
    I_E_MARK  = symbol(string(take([`O, `I], n)))      
    I_E_PORT = symbol_data  
    I_E_DATE = date_data         
    TRADE_COUNTRY = symbol_data  
    TRADE_MODE = symbol_data  
    TRANS_MODE = symbol_data  
    WRAP_TYPE = symbol_data 
    BUSI_TYPE = symbol_data 
    GROSS_WT =  double_data      
    NET_WT=  double_data       
    FEE_RATE =  double_data       
    INSUR_RATE =  double_data      
    OTHER_RATE=  double_data       
    TRAF_NAME = symbol_data  
    VOYAGE_NO = string_data  
    CONTAINER_NO=  double_data      
    CO_OWNER = symbol_data 
    DUTY_MODE = symbol_data  
    G_NO =  double_data       
    ITEM_NO =  double_data       
    UNIT = symbol_data  
    DEC_PRICE=  double_data       
    CURR = symbol_data  
    RMB_PRICE = take(1..n, n)    
    G_NAME = string_data  
    G_MODEL = string_data 
    DEC_PRICE_USD = take(1..n, n) 
    HS_CODE = symbol_data  
    GOODS_WEIGHT=  double_data      
    GOODS_PASS_TIME= date_data      
    FORM_PASS_TIME = date_data      
    TRADE_SOCIAL_CODE = symbol_data  
    TAX_REBATE_RATE=  double_data       
    TAX_CHARGE_RATE=  double_data       
    INFOTIME = take(date(bdate),n)        
    
    t = table(OID, ENTRY_ITEM_NO,ENTRY_ID, DECLARE_DATE,CUSTOM_MASTER,DECLARE_CODE,DECLARE_NAME,OWNER_CODE,OWNER_NAME,TRADE_CODE,TRADE_NAME,DISTINATE_PORT,
DISTRICT_CODE,I_E_MARK,I_E_PORT,I_E_DATE,TRADE_COUNTRY,TRADE_MODE,TRANS_MODE,WRAP_TYPE,BUSI_TYPE,GROSS_WT,NET_WT,FEE_RATE,INSUR_RATE, OTHER_RATE,TRAF_NAME,
VOYAGE_NO, CONTAINER_NO,CO_OWNER, DUTY_MODE, G_NO,ITEM_NO,UNIT,DEC_PRICE,CURR, RMB_PRICE, G_NAME,G_MODEL,DEC_PRICE_USD,HS_CODE, GOODS_WEIGHT,GOODS_PASS_TIME,
FORM_PASS_TIME,TRADE_SOCIAL_CODE,TAX_REBATE_RATE,TAX_CHARGE_RATE,INFOTIME)	

	return t
}

def write_data(begindate, enddate)
{
    bdate = begindate;
    primary_key = 0;
    do{
        t = create_table(bdate, 14400)
        bdate = bdate + 24*60*60;
        SZ_ENTRY_INFO=loadTable("dfs://db_SZ_ENTRY_INFO","SZ_ENTRY_INFO")
        SZ_ENTRY_INFO.append!(t)
        t=NULL
    }while(bdate<=enddate)
}
