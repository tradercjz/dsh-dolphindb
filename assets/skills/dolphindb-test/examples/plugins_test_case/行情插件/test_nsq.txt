@testing:case="test_nsq_getSubscriptionStatus_multithreading"
scm = schema(snapshot_sh_dummy).colDefs
enableTableShareAndPersistence(table=streamTable(1000000:0, scm[`name], scm[`typeInt]), tableName="snapshot_sh", cacheSize=1000000)
go
nsq::subscribe("snapshot",  "sh", snapshot_sh)
sleep(1000)
def multithreading_getSubscriptionStatus(){
	res = true
	for(i in 0..10){
		re = select * from nsq::getSubscriptionStatus() order by topicType
        res&&=size(re)==1
        res&&=(re[0][`topicType]=="(snapshot, sh)")
        res&&=(re[0][`startTime]!=NULL)
        res&&=(re[0][`endTime]==NULL)
        res&&=(re[0][`firstMsgTime]!=NULL)
        res&&=(re[0][`lastMsgTime]!=NULL)
        res&&=(re[0][`processedMsgCount]>0)
        res&&=(re[0][`lastErrMsg]==NULL)
        res&&=(re[0][`failedMsgCount]==0)
        res&&=(re[0][`lastFailedTimestamp]==NULL)
	}
	
	return res
}
