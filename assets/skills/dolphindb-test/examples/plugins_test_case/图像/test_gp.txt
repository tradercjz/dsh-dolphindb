@testing:case="test_gp_plot_style"

def  checktypes(d,WORK_DIR,prop,dt){	
	clear(WORK_DIR)
	data=char(d)
	re=gp::plot(data,"line",WORK_DIR+"/test.eps",prop)
	re=gp::plot(data,"line",WORK_DIR+"/test.png",prop)
	re=gp::plot(data,"line",WORK_DIR+"/test.jpeg",prop)
	sleep(3000)
	assert 1,exists(WORK_DIR+"/test.eps")
	assert 2,exists(WORK_DIR+"/test.jpeg")
	assert 3,exists(WORK_DIR+"/test.png")
	clear(WORK_DIR)
	data=short(d)
	re=gp::plot(data,"point",WORK_DIR+"/test.eps",prop)
	re=gp::plot(data,"point",WORK_DIR+"/test.png",prop)
	re=gp::plot(data,"point",WORK_DIR+"/test.jpeg",prop)
	sleep(3000)

		rm(WORK_DIR+"/test"+i+".eps")
	}
	if(exists(WORK_DIR+"/test"+i+".png")){

	}

		rm(WORK_DIR+"/test"+i+".jpeg")
	}

	n=100
	tabledata1=table(rand(10.0,n) as col1,rand(10.0,n) as col2)
	tabledata2=table(rand(10.0,n) as col1,rand(10.0,n) as col2)
	data=array(ANY,2)
	data[0]=tabledata1
	data[1]=tabledata2
	for(i in 0:30){
	re=gp::plot(data,"line",WORK_DIR)
	}
}
jobs=array(STRING)
for(i in 0:5){
	jobs.append!(submitJob("pgThread"+i+"","",pgThread,WORK_DIR+"/test"+i+".eps"))
	jobs.append!(submitJob("pgThread"+i+"","",pgThread,WORK_DIR+"/test"+i+".png"))
	jobs.append!(submitJob("pgThread"+i+"","",pgThread,WORK_DIR+"/test"+i+".jpeg"))
}
for(job in jobs){
	getJobReturn(job,1)
}
for(i in 0:5){
	assert 1,exists(WORK_DIR+"/test"+i+".eps")
	assert 2,exists(WORK_DIR+"/test"+i+".png")
	assert 3,exists(WORK_DIR+"/test"+i+".jpeg")
}
