@testing:case="binance"
def parseAggTrades(rawDatas){
    total = table(1000:0,`type`eventTime`code`id`price`quantity`firstId`lastId`tradeTime`marketMaker`currentTime,[SYMBOL,TIMESTAMP,SYMBOL,LONG,DOUBLE,DOUBLE,LONG,LONG,TIMESTAMP,BOOL,TIMESTAMP])
    for (rawData in rawDatas){
        rawDataDict = parseExpr(string(rawData[`msg])).eval()
        if (rawDataDict[`e]!=NULL){
            type=symbol([rawDataDict["e"]])
            eventTime=timestamp(long([rawDataDict["E"]]))
            code=symbol([rawDataDict["s"]])
            id=long([rawDataDict["a"]])
            price=double([rawDataDict["p"]])
            quantity=double([rawDataDict["q"]])
            firstId=long([rawDataDict["f"]])
            lastId=long([rawDataDict["l"]])
            tradeTime=timestamp(long([rawDataDict["T"]]))
            marketMaker=bool([rawDataDict["m"]])
            currentTime=[now()-28800000]
            res=table(type, eventTime, code, id, price, quantity, firstId, lastId, tradeTime, marketMaker, currentTime)
            total.append!(res)
        }
    }
    return total
}

colNames =`type`eventTime`code`id`price`quantity`firstId`lastId`tradeTime`marketMaker`currentTime
