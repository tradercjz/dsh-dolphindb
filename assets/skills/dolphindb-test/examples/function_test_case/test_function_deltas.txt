def checkDecimalArray(dtype, n, data, expectedData){
	actData=take(data, n)
	re = deltas(dtype(actData, 3))
	return eqObj(dtype(expectedData, 3), re, 3)
}

def checkDecimalSubarray(dtype, n, data){
	decimal_data = dtype(data, 3)
	range1 = rand(1..size(data), 1)[0]
	range2 = rand(range1..size(data), 1)[0]
	re = deltas(subarray(decimal_data, range1:range2))
	ex = dtype(deltas(subarray(data, range1:range2)), 3)
	return eqObj(ex, re, 2)
}

def checkDecimalBigarray(dtype1, n, data, dtype2){
	re_data = bigarray(dtype1, 0, n).append!(take(data, n))
	ex_data = bigarray(DOUBLE, 0, n).append!(take(data, n))
	re = deltas(re_data)
	ex = dtype2(deltas(ex_data), 3)
	return eqObj(ex, re, 1)
}

@testing:case="deltas_X_types"
def apply_types(input,types){
    return call{,input}:E(funcByName:E(types))
}
types=`bool`char`short`int`long`date`month`time`minute`second`datetime`timestamp`nanotime`nanotimestamp`float`double
for (len in [200,1024,2000]){
    vec=rand(100,len)
    narray=(-10..10).join([size(vec)-1,size(vec),size(vec)+1])
    for (n in narray){
        assert 1,apply_types(vec,types)-apply_types(move(vec,n),types)==call{deltas,,n}:E(apply_types(vec,types))
    }
    mat=vec$(len/8):8
    for (n in narray){
        assert 2,apply_types(mat,types)-apply_types(move(mat,n),types)==call{deltas,,n}:E(apply_types(mat,types))
    }
    t=table(apply_types(vec,types))
    assert 3, each(eq,values(deltas(t,n)),values(t-(move(t,n))))
}
