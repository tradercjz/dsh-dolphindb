def checkDecimalArray(dtype, n){
	data = rand(rand(-100..100, 10) join take(00i, 3), n)
	re = abs(dtype(data, 3))
	ex = dtype(abs(data), 3)
	return eqObj(ex, re)

def checkDecimalSubarray(dtype, n){
	data = rand(rand(-100..100, 10) join take(00i, 3), n)
	decimal_data = dtype(data, 3)
	range1 = rand(1..size(data), 1)[0]
	range2 = rand(range1..size(data), 1)[0]
	re = abs(subarray(decimal_data, range1:range2))
	ex = dtype(abs(subarray(data, range1:range2)), 3)
	return eqObj(ex, re)
}

	data = rand(rand(-100..100, 10) join take(00i, 3) join take([0.123, 0.456, -0.2154], 3) , n)

	re_data = bigarray(DECIMAL32(3), 0, n).append!(data)

	ex = dtype(abs(ex_data), 3)
