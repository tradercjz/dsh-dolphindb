@testing:case = "test_tmoving_excludedPeriod_prepare"
def change_time(time,excludedPeriod){
	cast_excludedPeriod = excludedPeriod$NANOTIME
	period_size = cast_excludedPeriod[1] - cast_excludedPeriod[0]
	
	origin_type = type(time[0])
	if(origin_type in [12,14]){
		cast_time = time$NANOTIMESTAMP
		time_minus = cast_time[nanotime(cast_time)<=cast_excludedPeriod[0]]
		time_over = cast_time[nanotime(cast_time)>=cast_excludedPeriod[1]]
	}else{
		cast_time = time$NANOTIME
		time_minus = cast_time[cast_time<=cast_excludedPeriod[0]]
		time_over = cast_time[cast_time>=cast_excludedPeriod[1]]
	}
	return ((time_minus join (time_over-period_size))$origin_type).sort()
}

@testing:case="test_tmoving_func_covarp_T_number"
dtypes = [int,long,short]
n = 2000
for(d in dtypes){
	for(dd in dtypes){
		for(ddd in dtypes){
			T = d(1..n)
			x = dd(1..n)
			y = rand(-100..100,n)
			window = ddd(2)
			re= tmoving(covarp, T, [x,y], window)
			ex = tmcovarp(T,x,y,window)
			assert 1,eqObj(re,ex)
			T = d(rand(1..5000,n).sort())
			re= tmoving(covarp, T, [x,y], window)
			ex = tmcovarp(T,x,y,window)
			assert 2,eqObj(re,ex)
			window = ddd(1000)
			re= tmoving(covarp, T, [x,y], window)
			ex = tmcovarp(T,x,y,window )
			assert 3,eqObj(re,ex)
			x = take(dd(),n)
			T = d(rand(1..100,n).sort())
			re= tmoving(covarp, T, [x,y], window)
			ex = tmcovarp(T,x,y,window)
			assert 4,eqObj(re,ex)
		}
	}
}
