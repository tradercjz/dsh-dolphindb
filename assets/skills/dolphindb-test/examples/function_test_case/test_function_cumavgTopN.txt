@testing:case="test_prepare_vector"
defg mavgTN_true(a, b,top_num,ascending,tiesMethod){
	x = array(type(a[0])).append!(a)
	z = b
	if(ascending)
		cur = z.size()-z.count()
	else 
		cur= 0
	//z = nullFill(z, min(z)-1 )
	if (tiesMethod=='oldest')
		return avg(x[isort(z,ascending)[cur:(cur+min(top_num,z.count() ))]] )
	else if(tiesMethod=='latest' ){
		x=x[size(x):0]
		z=z[size(z):0]
		return avg(x[isort(z,ascending)[cur:(cur+min(top_num,z.count() ))]] )
		}
	else if(tiesMethod=='all'){
		rank_z = rank(z,ascending)
		rank_z = rank_z[rank_z<top_num]
		get_number = count(rank_z)
		if(get_number>top_num)
			top_num_all = get_number
		else
			top_num_all = top_num
		return avg(x[isort(z,ascending)[cur:(cur+min(top_num_all,z.count() )) ]] )
		}
	else{throw('error')}
}

def my_cumavgTopN(a, b,top_num,ascending=true,tiesMethod='latest'){
	res = []
	for (i in 1..a.size()){
		res.append!(mavgTN_true( a[0:i], b[0:i],top_num,ascending,tiesMethod ) )	
	}
	return res
}

//---------------------------------    support matrix-------------------------------------------------------------

def my_cumavgTopN_matrix(a, b,top_num,ascending=true,tiesMethod='latest'){
	res = []
	for (i in 0..(a.cols()-1) ){
		a_0 =a[i]
		sub_res = []
		for (j in 1..a_0.size() ){
			sub_res.append!(mavgTN_true( a_0[0:j],b[0:j],top_num,ascending,tiesMethod ) )
			}
		res.append!(cast(sub_res,DOUBLE))	
	}
	return matrix(res)
}

//S is matrix
def my_cumavgTopN_Xmatrix_Smatrix(a, b,top_num,ascending=true,tiesMethod='latest'){
	res = []
	for (i in 0..(a.cols()-1) ){
		a_0 =a[i]
		b_0 = b[i]
		sub_res = []
		for (j in 1..a_0.size() ){
			sub_res.append!(mavgTN_true( a_0[0:j],b_0[0:j],top_num,ascending,tiesMethod ) )
			}
		res.append!(cast(sub_res,DOUBLE))	
	}
	return matrix(res)
}

re = select cumavgTopN(price,id,5,true) from pt where date>2021.05.01 context by sym
tmp_a = select * from pt where sym = `a and date>2021.05.01
prices_a = exec price from tmp_a
ids_a = exec id from tmp_a
ex_a = cumavgTopN(prices_a,ids_a,5,true)

tmp_b = select * from t where sym = `b and date>2021.05.01
prices_b = exec price from tmp_b
ids_b = exec id from tmp_b
ex_b= cumavgTopN(prices_b,ids_b,5,true)

ex = ex_a.append!(ex_b)
assert 1,each(eqObj,re.values()[0],ex)

@testing:case = "test_cumavgTopN_in_memory_table"
n = 3000
ids = 1..3000 
dates = take(2021.01.01..2021.10.01,n)
prices = rand(1000,n)
vals = rand(1000,n)
t = table(ids as id,dates as date,prices as price,vals as val)
re = select cumavgTopN(price,id,5,true) from t 
ex = cumavgTopN(prices,ids,5,true)
assert 1,each(eq,re.values()[0],ex)

re = select cumavgTopN(price,id,5,false) from t 
ex = cumavgTopN(prices,ids,5,false)
assert 2,each(eqObj,re.values()[0],ex)
