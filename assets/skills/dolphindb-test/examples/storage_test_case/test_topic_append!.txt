def createDB(dbName, tableName, t, days){
	if(existsDatabase(dbName)){
		dropDatabase(dbName)
	}
	db1 = database("",VALUE,days)
	db2=database("",HASH,[IPADDR,10])
	db = database(dbName,COMPO,[db1,db2], , "TSDB")
	db.createPartitionedTable(t,tableName,`cdate`cip, , `cdate)
}

def generateTable(n){
	basicTable= table(100:0,`id`cbool`cchar`cshort`cint`clong`cdate`cmonth`ctime`cminute`csecond`cdatetime`ctimestamp`cnanotime`cnanotimestamp`cfloat`cdouble`csymbol`cstring`cuuid`cip`cint128,[LONG,BOOL,CHAR,SHORT,INT,LONG,DATE,MONTH,TIME,MINUTE,SECOND,DATETIME,TIMESTAMP,NANOTIME,NANOTIMESTAMP,FLOAT,DOUBLE,SYMBOL,STRING,UUID,IPADDR,INT128])
	vid = 1..n
	vbool = rand([true,false],n)
	vchar = rand(-127..127,n)
	vshort = rand(-32767..32767,n)
	vint = rand(-214747..2143647,n)
	vlong = rand(-92237..922807,n)
	vdate = take(2012.06.12..2016.05.02,n) //first 
	vmonth = rand(2010.01M..2020.12M,n)
	vtime = rand(00:00:00.001..00:00:59.999,n)
	vminute = rand(01:00m..23:02m,n)
	vsecond = rand(01:58:59..12:02:01,n)
	vdatetime = rand(2018.01.01T01:58:59..2018.01.02T12:02:01,n)
	vtimestamp = rand(2018.01.01T01:58:59.001..2018.01.01T01:59:59.001,n)
	vnanotime = rand(01:58:59.000000001..01:58:59.000100000,n)
	vnanotimestamp = rand(2018.01.01T01:58:59.000000001..2018.01.01T01:58:59.000100001,n)
	vfloat = rand(1..10000,n)
	vdouble = rand(-19999..19999,n)
	vsymbol = take(`A`B`C`D`E,n) // second
	vstring = rand(`APPL`GOOG`MS`INTEL`BBA,n)
	vuuid=rand(uuid(),n)
	vip=rand(ipaddr("192.168.1."+string(1..200)),n)
	vint128=rand(int128(),n)
	insert into basicTable values(vid, vbool,vchar,vshort,vint,vlong,vdate,vmonth,vtime,vminute,vsecond,vdatetime,vtimestamp,vnanotime,vnanotimestamp,vfloat,vdouble,vsymbol,vstring,vuuid,vip,vint128)
	return basicTable
}

def createDatabase(dbName){
	if(existsDatabase(dbName)){
		dropDatabase(dbName)
	}
	db = database(dbName, VALUE, 1..30, , "TSDB")
	colNames = `id`ts`code`cbool`cchar`cshort`cint`clong`cdate`cmonth`ctime`cminute`csecond`cdatetime`ctimestamp`cnanotime`cnanotimestamp`cfloat`cdouble`csymbol`cstring`cdatehour`cint128`cuuid`cipaddr`cblob`ccomplex`cpoint`cdecimal32`cdecimal64
	colTypes = [INT, TIMESTAMP, SYMBOL, BOOL, CHAR, SHORT, INT, LONG, DATE, MONTH, TIME, MINUTE, SECOND, DATETIME, TIMESTAMP, NANOTIME, NANOTIMESTAMP, FLOAT, DOUBLE, SYMBOL, STRING, DATEHOUR, INT128, UUID, IPADDR, BLOB, COMPLEX, POINT, DECIMAL32(4), DECIMAL64(6)]
	dummy = table(1:0, colNames, colTypes)
	pt1 = db.createPartitionedTable(dummy, "pt1", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, ALL)
	pt2 = db.createPartitionedTable(dummy, "pt2", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, FIRST)
	pt3 = db.createPartitionedTable(dummy, "pt3", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, LAST)
	pt4 = db.createPartitionedTable(dummy, "pt4", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, ALL, [hashBucket{, 20}, ])
	pt5 = db.createPartitionedTable(dummy, "pt5", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, FIRST, [hashBucket{, 20}, ])
	pt6 = db.createPartitionedTable(dummy, "pt6", `id, {"cdate":"delta", "clong":"delta"}, `code`id`ts, LAST, [hashBucket{, 20}, ])
	dt1 = db.createTable(dummy, "dt1", {"cdate":"delta", "clong":"delta"}, `code`id`ts, ALL)
	dt2 = db.createTable(dummy, "dt2", {"cdate":"delta", "clong":"delta"}, `code`id`ts, FIRST)
	dt3 = db.createTable(dummy, "dt3", {"cdate":"delta", "clong":"delta"}, `code`id`ts, LAST)
}

def writeData(dbName, tableName, data){
	loadTable(dbName, tableName).append!(data)
}

colNames = `id`ts`code`cbool`cchar`cshort`cint`clong`cdate`cmonth`ctime`cminute`csecond`cdatetime`ctimestamp`cnanotime`cnanotimestamp`cfloat`cdouble`csymbol`cstring`cdatehour`cint128`cuuid`cipaddr`cblob`ccomplex`cpoint`cdecimal32`cdecimal64
