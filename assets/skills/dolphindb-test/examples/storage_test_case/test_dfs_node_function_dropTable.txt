def myDropTable(dbName){
	db = database(dbName)
	do{		
		tbNames  = exec name from tableLists where state==1
		for(tbName in tbNames){	
			print("try drop " + tbName)
			dropTable(db, tbName)
			update tableLists set state=2 where  name== tbName	
		}
		if((tableLists[`state]==2).all()){
			break;
		}
		
	}
	while(1)
}
