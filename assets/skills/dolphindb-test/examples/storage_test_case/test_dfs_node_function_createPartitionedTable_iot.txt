@testing:case="test_dfs_node_function_createPartitionedTable_iot_multiThreading_create_dropTable"
def createDB(dbName){	
	if(existsDatabase(dbName)){
	dropDatabase(dbName)
	}		
	dbDate = database(, HASH, [INT,5])
	dbx = database(, VALUE, 1..10)
	db = database(dbName, COMPO, [dbDate,dbx],engine="TSDB")
}

def writeDate(dbName,tableNames){
	for(tableName in tableNames){
		n=100
		ID=rand(100, n)
		dates=2017.08.07..2017.08.11
		date=rand(dates, n)
		x=rand(10, n)
		ticker = string(1..n);
		sym = rand(`MSFT`GOOG`FB`ORCL`IBM,n);
		t=table(ID, date, x,ticker,sym)
		loadTable(dbName,tableName).append!(t)
	}
}
def createPartTables(dbName,tableNames){
	db=database(dbName)
	t=table(10:0,`ID`date`x`ticker`sym,[INT,DATE,INT,STRING,SYMBOL])	
	for(tableName in tableNames){
		if(!existsTable(dbName,tableName)){
			m=rand(0 1,1)[0]
			if(m==0){
				db.createPartitionedTable(t, tableName, `ID`x,{date:"delta"},`date)
			}
			else if(m==1){
				db.createPartitionedTable(t, tableName, `ID`x,{date:"delta"},`sym`ticker`date)
			}
		}
	}
}

def createTables(dbName,tableNames){
	db=database(dbName)
	t=table(10:0,`ID`date`x`ticker`sym,[INT,DATE,INT,STRING,SYMBOL])	
	for(tableName in tableNames){
		if(!existsTable(dbName,tableName)){
			m=rand(0 1,1)[0]
			 if(m==0){
				db.createTable(t, tableName,{date:"delta"},`date)
			}
			else if(m==1){
				db.createTable(t, tableName,{date:"delta"},`sym`ticker`date)
			}	
		}
	}
}

def dropPartTable(dbName,tableNames){
	db=database(dbName)
	tableName=rand(getTables(db),1)[0]
	if(existsTable(dbName,tableName)){
		db.dropTable(tableName)
	}
}

def testcreate(dbName,tableNames){
	for(i in 0:2){
		if(existsDatabase(dbName)){
			createPartTables(dbName,tableNames[0:10])
			createTables(dbName,tableNames[10:20])
			tableName=rand(tableNames,(tableNames.size()/2))
			writeDate(dbName,tableName)
			for(i in 0:20){
			dropPartTable(dbName,tableNames)
			}
		}
	}
}
createDB("dfs://test_db1")

createDB("dfs://test_db3")
createDB("dfs://test_db4")
job1=submitJob("test1","",testcreate,"dfs://test_db1","pt"+string(1..20))
