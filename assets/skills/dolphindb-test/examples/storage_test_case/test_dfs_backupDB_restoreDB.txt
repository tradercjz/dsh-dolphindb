def checkDict(ex, re, excludeKey){
	ex_keyNum = ex.keys().size()
	re_keysNum = re.keys().size()
	if(ex_keyNum!=re_keysNum){
		return false
	}
	comparison = array(BOOL, 0, 10)
	for(k in ex.keys()){
		if(k!=excludeKey){
			if(form(ex[k])==6){
				tmp = each(eqObj, values(ex[k]), values(re[k])).flatten().all()
			}else{
				tmp = eqObj(ex[k], re[k])
			}
			comparison.append!(tmp)
		}
	}
	return all(comparison)
}
