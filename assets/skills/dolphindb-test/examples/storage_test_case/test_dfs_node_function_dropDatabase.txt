print(i)

dropDatabase(dbPath)
}
n=600000
db1 = database("", VALUE, 2020.02.01 .. 2020.02.10)
db2 = database("", HASH, [STRING, 10])
db = database(dbPath, COMPO, [db1, db2], , "TSDB")
t = table(timestamp(take(2020.02.01 .. 2020.02.10,n))+take(i,n) as timestamp, take("id" +string(1..50), n) as id, take(1.0, n) as value)
tem = t
pt = db.createPartitionedTable(t, "pt", `timestamp`id, , `id`timestamp).append!(t)
pt1 = db.createPartitionedTable(t, "pt1", `timestamp`id, , `id`timestamp).append!(t)
pt2 = db.createPartitionedTable(t, "pt2", `timestamp`id, , `id`timestamp).append!(t)
re=select * from loadTable(dbPath, `pt) order by timestamp, id
re1=select * from loadTable(dbPath, `pt1) order by timestamp, id
re2=select * from loadTable(dbPath, `pt2) order by timestamp, id
