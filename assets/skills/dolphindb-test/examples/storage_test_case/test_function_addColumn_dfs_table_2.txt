@testing:case = "test_function_addcolumn_dfs_table_range_partition_onecol_decimal32"
def func1(table_data, insertdata1){
	arr = array(BOOL, 0, 100)
	t0 = table_data
	if(existsDatabase("dfs://rangedb"))
		dropDatabase("dfs://rangedb")
	db=database("dfs://rangedb",RANGE,  0 51 101,, `TSDB)
	pt = db.createPartitionedTable(t0, `pt, `id,,`id)
	addColumn(pt,"add1",DECIMAL32(3));
	pt = loadTable(db, `pt)
	pt.append!(t0);
	re1 = select * from pt order by ID
	ex1 = table_data
	update ex1 set add1 = decimal32(,3)
	ex1 = select * from ex1 order by ID
	arr.append!(eqObj(re1.values(), ex1.values()))
	insert1 = insertdata1
	update insert1 set add1 = decimal32(-9,3)
	pt.append!(insert1)
	pnodeRunAllNodes(flushOLAPCache)
	re2 = select * from pt order by ID
	ex1.append!(insert1)
	ex2 = select * from ex1 order by ID
	arr.append!(eqObj(re2.values(), ex2.values()))
	re3 = select name, typeString, typeInt, extra, comment from schema(pt)[`colDefs]
	ex3 = select name, typeString, typeInt, extra, comment from schema(ex2)[`colDefs]
	arr.append!(eqObj(re3.values(), ex3.values()))

	addColumn(pt,"add2",DECIMAL64(3));
	pt = loadTable(db, `pt)
	update insert1 set add2 = decimal64(9,3)
	pt.append!(insert1)
	pnodeRunAllNodes(flushOLAPCache)
	sleep(500)
	re4 = select * from pt order by ID
	update ex2 set add2 = decimal64(,3)
	ex2.append!(insert1)
	ex4 = select * from ex2 order by ID
	arr.append!(eqObj(re4.values(), ex4.values()))
	re5= select name, typeString, typeInt, extra, comment from schema(pt)[`colDefs]
	ex5 = select name, typeString, typeInt, extra, comment from schema(ex4)[`colDefs]
	arr.append!(eqObj(re5.values(), ex5.values()))
	return arr
}
tables = [alltypetable_0, alltypetable_100, alltypetable_1024, alltypetable_65535]
assert 1, each(func1{,insertdata1}, tables)
assert 2, func1(wideTable,insertwidedata1)
