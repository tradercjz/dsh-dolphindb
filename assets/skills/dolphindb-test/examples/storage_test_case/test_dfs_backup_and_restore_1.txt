@testing:case="function_backup_restore_single_value"
single_db = "dfs://backup_single_value"
single_tb = "valuepart1"
createValueDB(single_db, single_tb, tableSchema)
writeData(single_db, single_tb) 
tb = database(single_db).loadTable(single_tb)
ct1 = select count(*) from tb
assert 1, ct1.count[0] == 5000
backup(workdir_backup, <select * from tb>)
writeData(single_db, single_tb) 
ct1 = select count(*) from tb
assert 2, ct1.count[0] == 10000
restore(workdir_backup, single_db, single_tb, "%")
ct1 = select count(*) from tb
assert 3, ct1.count[0] == 5000
restore(workdir_backup, single_db, single_tb, "%", true)
ct1 = select count(*) from tb
assert 4, ct1.count[0] == 5000

single_tb_copy = "valuepart1_copy"
createValueDB(single_db, single_tb_copy, tableSchema)
single_tb_copy_case =  database(single_db).loadTable(single_tb_copy)
restore(workdir_backup, single_db, single_tb, "%",false, single_tb_copy_case)
ct1 = select count(*) from single_tb_copy_case
assert 5, ct1.count[0] == 5000
restore(workdir_backup, single_db, single_tb, "%",true, single_tb_copy_case)
ct1 = select count(*) from single_tb_copy_case
assert 6, ct1.count[0] == 10000
dropDatabase(single_db)
try{
	rmdir(workdir_backup,true)
}catch(ex){print ex}
