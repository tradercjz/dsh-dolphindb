def getVolumeId(replicas){
	result = array(LONG, 0)
	replica = replicas.split(",")
	for(rep in replica){
		re = rep.split(":")
		result.append!(long(re[5]))
	}
	return [result]
}

def getPath(path){
	p = path.split("/")
	for(i in 0:p.size()){
		if(p[i] == "CHUNKS"){
			break	
		}	
	}
	return concat(p[:i], "/")
}

def checkVolumeId(){
	res1 =  select chunkId, replicas from rpc(getControllerAlias(),getClusterChunksStatus)
	res2 = select chunkId, path from pnodeRunAllNodes(getAllChunks)
	res = select each(getVolumeId,replicas) as f1 , chunkId, each(getPath,path) as f2 from res1 inner join res2 on res1.chunkId = res2.chunkId
	ext = select volumeId as volumeId, f1, f2 from getClusterVolumeUsage() as t1 inner join res on t1.volumePath = res.f2
	return each(def(x,y):x in y, ext.volumeId, ext.f1).all()
}
