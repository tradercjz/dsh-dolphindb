@testing:case="test_function_createReactiveStateEngine_keyPurgeFilter_delete_part_cumpercentile"
n = 1000
timestamp=temporalAdd(2012.01.01T12:23:23.125, 1..n, "s")
sym=take(`AA, n) 
askSeqNum=take(`name1, n) 
Side=loop(take{, 100}, "BB"+string(1..10)).flatten()
price=double(rand(100, n))
qty=double(rand(100, n))
t1 = table(timestamp, sym, askSeqNum, Side, price, qty)
dummy = table(1:0, `timestamp`sym`askSeqNum`Side`price`qty, [TIMESTAMP, SYMBOL, SYMBOL, SYMBOL, DOUBLE, DOUBLE])
outputTable = table(1000:0, `sym`askSeqNum`Side`updatetime`factor1`factor2`factor3`factor4`factor5, [SYMBOL, SYMBOL, SYMBOL, TIMESTAMP, DOUBLE, DOUBLE, DOUBLE, DOUBLE, DOUBLE])
engine = createReactiveStateEngine(name = "test", metrics = [<timestamp as updateTime>, <cumpercentile(price, 25, "linear")>, <cumpercentile(price, 35, "lower")>, <cumpercentile(price, 45, "higher")>, <cumpercentile(price, 55, "nearest")>, <cumpercentile(price, 65, "midpoint")>], dummyTable = dummy, outputTable = outputTable, keyColumn = `sym`askSeqNum`Side, keyPurgeFilter = <updateTime<temporalAdd(2012.01.01T12:23:23.125, n/2, "s")>, keyPurgeFreqInSecond = 3)
sleep(4000)
engine.append!(t1)
dummy.append!(t1)

re1= (exec numGroups from getStreamEngineStat().ReactiveStreamEngine where name = `test)[0]
assert 5, re1==6
ex = select sym, askSeqNum, Side, timestamp as updateTime, cumpercentile(price, 25, "linear") as factor1, cumpercentile(price, 35, "lower") as factor2, cumpercentile(price, 45, "higher") as factor3, cumpercentile(price, 55, "nearest") as factor4, cumpercentile(price, 65, "midpoint") as factor5 from dummy context by sym, askSeqNum, Side order by sym, askSeqNum, Side, updateTime
re = select * from outputTable order by sym, askSeqNum, Side, updateTime
assert 6, each(eqObj, re.values(), ex.values())

delete from dummy where Side in "BB"+string(1..4)

//write again
timestamp=temporalAdd(2012.01.01T12:23:23.125, (n+1)..(2*n), "s")
sym=take(`AA, n) 
askSeqNum=take(`name1, n) 
Side=loop(take{, 100}, "BB"+string(1..10)).flatten()
price=double(rand(100, n))
qty=double(rand(100, n))
t2 = table(timestamp, sym, askSeqNum, Side, price, qty)
engine.append!(t2)
dummy.append!(t2)

re = select * from outputTable where updateTime>=temporalAdd(2012.01.01T12:23:23.125, n+1, "s") order by sym, askSeqNum, Side, updateTime
re1 = select * from re where Side in "BB"+string(1..4)
re2 = select * from re where in(Side, "BB"+string(1..4))==false
ex = select sym, askSeqNum, Side, timestamp as updateTime, cumpercentile(price, 25, "linear") as factor1, cumpercentile(price, 35, "lower") as factor2, cumpercentile(price, 45, "higher") as factor3, cumpercentile(price, 55, "nearest") as factor4, cumpercentile(price, 65, "midpoint") as factor5 from dummy context by sym, askSeqNum, Side order by sym, askSeqNum, Side, updateTime
