@testing:case="test_function_createCrossSesctionalAggregator_outputTable_dfs_table"
share streamTable(10:0,`time`sym`price`qty,[TIMESTAMP,SYMBOL,DOUBLE,INT]) as trades
outputTable = table(1:0, `time`avgPrice`sumqty`Total, [TIMESTAMP,DOUBLE,INT,DOUBLE])

dbName = "dfs://test_CrossSesctionalAggregator"
if(existsDatabase(dbName)){
 dropDatabase(dbName)
}
db =database(dbName,VALUE,2018.10.01..2018.10.31)
db.createPartitionedTable(outputTable,`pt,`time)
db.createTable(outputTable,`dt)

tradesCrossAggregator = createCrossSectionalAggregator("CrossSectionalDemo", <[avg(price), sum(qty), sum(price*qty)]>, trades, loadTable(dbName,"pt"), `sym, `perRow,,false,`time)
tradesCrossAggregator1 = createCrossSectionalAggregator("CrossSectionalDemo1", <[avg(price), sum(qty), sum(price*qty)]>, trades, loadTable(dbName,"dt"), `sym, `perRow,,false,`time)
tradesCrossAggregator2 = createCrossSectionalAggregator("CrossSectionalDemo2", <[avg(price), sum(qty), sum(price*qty)]>, trades, outputTable, `sym, `perRow,,false,`time)

topic1 = subscribeTable(,"trades","tradesCrossAggregator",-1,append!{tradesCrossAggregator},true)
topic2 = subscribeTable(,"trades","tradesCrossAggregator1",-1,append!{tradesCrossAggregator1},true)
topic3 = subscribeTable(,"trades","tradesCrossAggregator2",-1,append!{tradesCrossAggregator2},true)

def writeData(n){
  timev  = 2018.10.01T01:01:01.001 +1..n
  symv   = take(`A`B`C, n)
  pricev = 1..n
  qtyv   = 1..n
  insert into trades values(timev, symv, pricev,qtyv)
}
writeData(100)
sleep(10000)

checkStreamingSubWorkers(topic1, trades.size())
checkStreamingSubWorkers(topic2, trades.size())
checkStreamingSubWorkers(topic3, trades.size())

re1 = select * from loadTable(dbName,"pt") order by time
re2 = select * from loadTable(dbName,"dt") order by time
ex = select * from outputTable order by time
assert 1,eqObj(re1.values(),ex.values())
assert 2,eqObj(re2.values(),ex.values())

unsubscribeTable(, "trades", "tradesCrossAggregator")
unsubscribeTable(, "trades", "tradesCrossAggregator1")
unsubscribeTable(, "trades", "tradesCrossAggregator2")
dropAggregator("CrossSectionalDemo")
dropAggregator("CrossSectionalDemo1")
dropAggregator("CrossSectionalDemo2")

tradesCrossAggregator1 = NULL
tradesCrossAggregator2 = NULL
