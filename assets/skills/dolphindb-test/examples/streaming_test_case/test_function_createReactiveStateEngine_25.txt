def getRowfuncCalculate(rowfunc, colName){
	//return rowfunc(colName)
	return parseExpr(string(rowfunc)+"("+colName+")")
}

@state
def getRowFuncSlice(rowfunc, colName){
	//return rowfunc(colName[2:5])
	return parseExpr(string(rowfunc)+"("+colName+"[2:5])")
}

def checkReactiveStateRowFunc(rowfunc, dtype1, dtype2, dtype3, data){
	try{dropStreamEngine("reactive")}catch(ex){print(ex)}
	dummy=table(1000:0, `sym`timestampv`high`low`close`price, [SYMBOL, TIMESTAMP, INT, INT, INT, dtype1])
	output=table(1000:0, `sym`factor1, [SYMBOL, dtype2])
	//engine=createReactiveStateEngine(name="reactive", metrics=[<call(getRowfuncCalculate, rowfunc, Price)>], dummyTable=dummy, outputTable=output, keyColumn="sym")
	m = getRowfuncCalculate(rowfunc, "price")
	engine=createReactiveStateEngine(name="reactive", metrics=m, dummyTable=dummy, outputTable=output, keyColumn="sym")
	dummy.append!(data)
	engine.append!(dummy)
	re=select * from output
	ex=select sym, dtype3(rowfunc(price)) as factor1 from dummy 
	return each(eqObj, re.values(), ex.values(), 6)
}
