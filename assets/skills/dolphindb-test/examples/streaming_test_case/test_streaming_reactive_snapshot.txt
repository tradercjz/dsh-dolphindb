@testing:case="test_streaming_aggregator_snapshotIntervalInMsgCount_ReactiveStateEngine_ema"
DIR = WORK_DIR+"/snapshotDir"
mkdir(DIR)
share streamTable(10000:0,`time`sym`price`id, [TIMESTAMP,SYMBOL,INT,INT]) as trades
outputTable = table(1000:0, `sym`factor1, [STRING,DOUBLE])
rengine = createReactiveStateEngine("test", <ema(price, 40)>, trades, outputTable, "sym",snapshotDir=DIR, snapshotIntervalInMsgCount=100)
topicName1=subscribeTable(server="", tableName="trades", actionName="rengine",offset= 0, handler=appendMsg{rengine}, msgAsTable=true, handlerNeedMsgId=true)
outputTable1 = table(1000:0, `sym`factor1, [STRING,DOUBLE])
rengine1 = createReactiveStateEngine("test1", <ema(price, 40)>, trades, outputTable1, "sym")
n=500
timev=timestamp(1..n)
symv = take(`AA`BB`CC, n)
pricev = 0..(n-1)
id = take(-1, n)
insert into trades values(timev, symv, pricev, id)
insert into rengine1 values(timev, symv, pricev, id)
waitStreamingEngine(`test, trades.size())
assert 1,eqObj(getSnapshotMsgId(rengine),499)
assert 2,eqObj(outputTable.values(),outputTable1.values())
go;

unsubscribeTable(, "trades", "rengine")
dropAggregator("test")
rengine =NULL

n=500
timev=timestamp(501..1000)
symv = take(`AA`BB`CC, n)
pricev = rand(100,n)
id = take(-1, n)
insert into trades values(timev, symv, pricev, id)
insert into rengine1 values(timev, symv, pricev, id)

rengine = createReactiveStateEngine("test", <ema(price, 40)>, trades, outputTable, "sym",snapshotDir=DIR, snapshotIntervalInMsgCount=100)
ofst = getSnapshotMsgId(rengine)
v = getAggregatorStat().ReactiveStreamEngine
assert 3,eqObj(v[`snapshotDir][0],DIR)
assert 4,eqObj(v[`snapshotInterval][0],100)
assert 5,eqObj(v[`snapshotMsgId][0],499)
go;

topicName2=subscribeTable(server="", tableName="trades", actionName="rengine",offset=ofst+1, handler=appendMsg{rengine}, msgAsTable=true, handlerNeedMsgId=true)
waitStreamingEngine(`test, n)
assert 6,eqObj(outputTable.values(),outputTable1.values())
go;

unsubscribeTable(, "trades", "rengine")
dropAggregator("test")
dropAggregator("test1")
rengine =NULL
rengine1 =NULL
rmdir(DIR,true)

	for(i in 1:N){
